1//! Layer-1 text layout (line wrapping) on top of the flux-text Layer-0 shaper.
2//!
3//! The **flux-text** sibling — mirrored by [`flux_text`] — deliberately stops
4//! at single-run shaping. The "API layers" note in `<flux-text/text.h>`
5//! draws the line explicitly:
6//!
7//! - **Layer 0** (implemented): shape / measure / draw one contiguous run of
8//!   one style. "Sufficient for labels and for callers that do their own
9//!   line/paragraph composition."
10//! - **Layer 1** (reserved `flux_text_layout`, "Not yet defined"): a
11//!   retained, cached layout object built from multiple styled runs with
12//!   line wrapping.
13//!
14//! This crate is a pure-Rust Layer-1 helper: it builds greedy word-wrap on
15//! top of [`flux_text::Text::measure`], so the engine boundary stays clean
16//! while consumers (and a future C `flux_text_layout`) have something to use
17//! today. Keeping it out of the `flux-text` crate means the sibling's Rust
18//! surface mirrors only Layer-0 — paragraph composition does not become part
19//! of flux-text's API contract until the engine actually defines Layer-1.
20//!
21//! Scope: single-style line wrapping with `\n` hard breaks and CJK
22//! per-character break opportunities. No UAX#14 break iteration, no multi-run
23//! paragraphs, no justification — those belong in a fuller layout engine
24//! (lens, or the future `flux_text_layout`).
25
26#![deny(rust_2018_idioms)]

30/// Inter-script auto-space, in EM units. Mirrors the same gap the flux-text
31/// shaper inserts at CJK↔non-CJK run boundaries (see `txt_run_autospace_em`
32/// in libflux's `text/layout.c`): a ¼-em breath so "你好hello" does not render
33/// glued. Standard CJK typography (CSS `text-autospace`, ctex/xeCJK) treats
34/// this as the default. `wrap` accounts for it so its line widths match what
35/// the shaper actually produces when the line is later drawn.
36const INTERSCRIPT_GAP_EM: f32 = 0.25;

38/// One visual line produced by [`wrap`]. The byte range `[lo, hi)` is in the
39/// original input's coordinates; `metrics` is the shaped extent of that
40/// substring under the style passed to `wrap`. Lines tile the input exactly
41/// once: `lines[i].hi == lines[i+1].lo` and the final `hi` is the input
42/// length (or the end of the last non-whitespace run).
43///
44/// Callers draw each line with [`Text::draw`] using `&text[line.lo..line.hi]`
45/// and the original style, advancing `y` by their chosen line stride between
46/// lines. Caret / hit-test mapping uses [`Text::x_for_byte`] /
47/// [`Text::byte_for_x`] on the same substring, with the byte offset first
48/// rebased into the line's local coordinates.
49#[derive(Debug, Clone, Copy, PartialEq)]
50pub struct WrappedLine {
51    pub lo: usize,
52    pub hi: usize,
53    pub metrics: Metrics,
54}

56/// Greedy word-wrap `input` to fit `max_width` (logical pixels), returning
57/// one [`WrappedLine`] per visual line. Each atom is measured as **raw
58/// source** at `style` — the convenience for callers whose drawn
59/// representation *is* the raw bytes (labels, plain text). Callers whose
60/// drawn representation differs from the raw source (hidden inline markup,
61/// per-run weight/family) should use [`wrap_with`] and supply the drawn-width
62/// measure themselves.
63pub fn wrap(text: &Text, input: &str, style: &Style, max_width: f32) -> Vec<WrappedLine> {
64    wrap_with(text, input, style, max_width, |r| {
65        text.measure(&input[r], style).width
66    })
67}

69/// Like [`wrap`], but each atom's width is supplied by `measure` rather than
70/// shaped from the raw source. `measure(byte_range)` returns the drawn width
71/// of `&input[byte_range]`, letting the caller account for a representation
72/// that differs from the raw bytes — inline markup whose markers are hidden,
73/// marks that change weight or family, … This keeps the Layer-1 crate free of
74/// any markup model: it stays a pure line-breaker, and all width-affecting
75/// representation choices live in the caller. It is also the per-atom cost
76/// hook a future optimal-breaker (Knuth-Plass) will consume.
77///
78/// The algorithm is greedy first-fit, built **on top of** the Layer-0 shaper
79/// `text`: `measure` is invoked once per atom (O(n)), and per-line widths are
80/// arithmetic over those widths. (An earlier prefix-remeasure implementation
81/// was O(n²) in atoms per line; the shaper's glyph cache let it survive, but
82/// a few-thousand-atom line made it visibly laggy.)
83///
84/// There is no UAX#14 break-iterator; break opportunities are:
85///
86/// - after each whitespace run (the whitespace attaches to the preceding
87///   line and is dropped when starting a new line),
88/// - between adjacent CJK ideographs (CJK text wraps without spaces),
89/// - at every `\n` (treated as a forced hard break),
90/// - **last resort:** if a single atom is wider than `max_width`, it is
91///   placed on its own line and overflows — the shaper cannot split a single
92///   glyph cluster. Long URLs / unbreakable strings will visually extend
93///   past `max_width`.
94///
95/// Empty input returns a single line at `(0, 0)` so callers can still
96/// position a caret. Whitespace-only / newline-only input likewise returns
97/// one empty line covering `[0, 0)`.
98///
99/// Width approximation: each inter-atom whitespace byte contributes the
100/// measured width of a single ASCII space (`base`). Tab and multi-space runs
101/// are therefore treated as N spaces. This matches typical editor behaviour
102/// (where tabs already render as a fixed number of spaces) and stays inside
103/// the Layer-1 "no full BiDi across the whole line" budget.
104///
105/// Inter-script auto-space: at a CJK↔non-CJK atom boundary with no
106/// separator, a ¼-em gap (`INTERSCRIPT_GAP_EM`) is added to mirror the
107/// flux-text shaper, which inserts the same gap when the line is drawn.
108/// Without it wrap would under-count and pack a line that overflows on
109/// screen. A boundary already covered by whitespace is left alone (no
110/// doubling).
111pub fn wrap_with(
112    text: &Text,
113    input: &str,
114    base: &Style,
115    max_width: f32,
116    measure: impl Fn(std::ops::Range<usize>) -> f32,
117) -> Vec<WrappedLine> {
118    if input.is_empty() {
119        return vec![WrappedLine {
120            lo: 0,
121            hi: 0,
122            metrics: text.measure("", base),
123        }];
124    }
125
126    let atoms = split_atoms(input);
127    if atoms.is_empty() {
128        return vec![WrappedLine {
129            lo: 0,
130            hi: 0,
131            metrics: text.measure("", base),
132        }];
133    }
134
135    // Pre-measure every atom once via the caller-supplied measure. This is
136    // the only per-atom shaping the algorithm does; everything below is
137    // arithmetic over those widths.
138    let atom_w: Vec<f32> = atoms.iter().map(|a| measure(a.start..a.end)).collect();
139
140    // Kinsoku classes per atom: whether it must not begin a line (no_start)
141    // or must not end one (no_end). A soft break between atoms idx-1 and idx
142    // is legal only when atoms[idx] may begin a line and atoms[idx-1] may
143    // end one; otherwise the break is forbidden and the atom is force-fitted.
144    let no_start: Vec<bool> = atoms.iter().map(|&a| atom_no_start(input, a)).collect();
145    let no_end: Vec<bool> = atoms.iter().map(|&a| atom_no_end(input, a)).collect();
146
147    // Width of one ASCII space — used to approximate the cost of the
148    // whitespace run between two adjacent atoms. Measuring each actual
149    // inter-atom whitespace run separately would re-introduce O(n²).
150    // Tab and multi-space runs collapse to N spaces, which matches how
151    // most editors and Layer-1 widgets render them anyway.
152    let space_w = text.measure(" ", base).width.max(0.0);
153
154    // Line metrics: height/baseline come from a base-style shape (font
155    // metrics, unaffected by marks); width is the caller's drawn width for
156    // the line's source range, so the returned metric matches what gets
157    // painted.
158    let line_metrics = |lo: usize, hi: usize| {
159        let mut m = text.measure(&input[lo..hi], base);
160        m.width = measure(lo..hi);
161        m
162    };
163
164    let mut lines: Vec<WrappedLine> = Vec::new();
165
166    // Index of the first atom on the current line, plus the byte end of
167    // the last atom that fit. The first atom on a line is always placed
168    // unconditionally (even if it overflows) so the loop terminates on a
169    // single atom wider than max_width.
170    let mut line_first = 0usize;
171    let mut line_w = atom_w[0];
172    let mut line_last_end = atoms[0].end;
173
174    let mut idx = 1usize;
175    while idx < atoms.len() {
176        // The gap between atoms idx-1 and idx (whitespace width + the
177        // shaper-mirrored inter-script auto-space), and whether it is a
178        // forced `\n` break. Shared with the optimal strategy via
179        // `gap_cost` so the inter-script rule has one source of truth.
180        let (gap_w, hard_break) = gap_cost(input, &atoms, idx, space_w, base.size_px);
181
182        if hard_break {
183            // `\n` forces a line break regardless of width.
184            let lo = atoms[line_first].start;
185            let hi = line_last_end;
186            lines.push(WrappedLine {
187                lo,
188                hi,
189                metrics: line_metrics(lo, hi),
190            });
191            line_first = idx;
192            line_w = atom_w[idx];
193            line_last_end = atoms[idx].end;
194        } else if line_w + gap_w + atom_w[idx] <= max_width || no_start[idx] || no_end[idx - 1] {
195            // Extend the line — either the atom fits, or kinsoku forbids
196            // breaking here: atoms[idx] is a no-start char (。，、）》」…)
197            // that must not begin a line, or atoms[idx-1] is a no-end
198            // opening bracket that must not end one. In the kinsoku case
199            // the line overflows rather than stranding punctuation at a
200            // line edge.
201            line_w += gap_w + atom_w[idx];
202            line_last_end = atoms[idx].end;
203        } else {
204            // Overflow, and breaking here is kinsoku-legal — close the
205            // current line and start a new one at this atom. Whitespace
206            // before the new line is dropped.
207            let lo = atoms[line_first].start;
208            let hi = line_last_end;
209            lines.push(WrappedLine {
210                lo,
211                hi,
212                metrics: line_metrics(lo, hi),
213            });
214            line_first = idx;
215            line_w = atom_w[idx];
216            line_last_end = atoms[idx].end;
217        }
218        idx += 1;
219    }
220
221    // Finalize the trailing line.
222    let lo = atoms[line_first].start;
223    let hi = line_last_end;
224    lines.push(WrappedLine {
225        lo,
226        hi,
227        metrics: line_metrics(lo, hi),
228    });
229
230    lines
231}

233// ----------------------------------------------------------------------
234// Knuth-Plass optimal line breaking
235// ----------------------------------------------------------------------
236//
237// The full paragraph-level DP from Knuth & Plass (*Digital Typography*):
238// minimise total demerit = Σ badness(line) + Σ adjacent-class demerit,
239// where each line is classified into a fitness class by how far it falls
240// short of `max_width`. The class term is what makes a K-P paragraph look
241// *even* — it penalises a tight line sitting next to a loose one — and is
242// the part a plain sum-of-shortfalls DP (which collapses to "fill each
243// line", ≈ greedy) misses.
244//
245// It consumes the same caller-supplied `measure` and the same `gap_cost`
246// rule as the greedy path, so it stays consistent with the shaper and with
247// the drawn representation. For ragged-right body text its edge over greedy
248// is real but subtle (most visible on dense CJK, where per-character break
249// opportunities multiply, and where greedy would strand a short final
250// line); it pays off fully once justification is in scope.
251
252/// Knuth-Plass optimal wrap — like [`wrap_with`] but choosing break points
253/// to minimise whole-paragraph demerit (badness + fitness-class evenness)
254/// rather than filling each line greedily. Same `measure` contract, same
255/// [`WrappedLine`] output, so it is a drop-in replacement for [`wrap_with`].
256/// See the "Knuth-Plass optimal line breaking" notes above for when it pays
257/// off. Each atom is measured as raw source at `style`; callers whose drawn
258/// representation differs from the raw bytes should use
259/// [`wrap_optimal_with`].
260pub fn wrap_optimal(text: &Text, input: &str, style: &Style, max_width: f32) -> Vec<WrappedLine> {
261    wrap_optimal_with(text, input, style, max_width, |r| {
262        text.measure(&input[r], style).width
263    })
264}

266/// Knuth-Plass optimal wrap with a caller-supplied measure. See
267/// [`wrap_with`] for the `measure` contract and [`wrap_optimal`] for the
268/// strategy. Falls back to [`wrap_with`] (greedy) for inputs above
269/// `KP_GREEDY_FALLBACK_ATOMS` atoms, where the O(n²·C) DP would risk
270/// dropping a frame on a megabyte paste.
271pub fn wrap_optimal_with(
272    text: &Text,
273    input: &str,
274    base: &Style,
275    max_width: f32,
276    measure: impl Fn(std::ops::Range<usize>) -> f32,
277) -> Vec<WrappedLine> {
278    if input.is_empty() {
279        return vec![WrappedLine {
280            lo: 0,
281            hi: 0,
282            metrics: text.measure("", base),
283        }];
284    }
285    let atoms = split_atoms(input);
286    if atoms.is_empty() {
287        return vec![WrappedLine {
288            lo: 0,
289            hi: 0,
290            metrics: text.measure("", base),
291        }];
292    }
293    if atoms.len() > KP_GREEDY_FALLBACK_ATOMS {
294        return wrap_with(text, input, base, max_width, measure);
295    }
296
297    let em = base.size_px.max(1.0);
298    let space_w = text.measure(" ", base).width.max(0.0);
299
300    let atom_w: Vec<f32> = atoms.iter().map(|a| measure(a.start..a.end)).collect();
301    // Kinsoku classes (see `wrap_with`). The DP only places line-ends at
302    // break points kinsoku allows, so an optimal solution never strands a
303    // no-start char at a line start or a no-end bracket at a line end.
304    let no_start: Vec<bool> = atoms.iter().map(|&a| atom_no_start(input, a)).collect();
305    let no_end: Vec<bool> = atoms.iter().map(|&a| atom_no_end(input, a)).collect();
306    let mut gap_w = Vec::with_capacity(atoms.len() - 1);
307    let mut gap_hard = Vec::with_capacity(atoms.len() - 1);
308    for k in 0..atoms.len() - 1 {
309        let (w, hard) = gap_cost(input, &atoms, k + 1, space_w, base.size_px);
310        gap_w.push(w);
311        gap_hard.push(hard);
312    }
313
314    // Prefix sums for O(1) line-width lookups. PA[k] = Σ atom_w[0..k];
315    // PG[k] = Σ gap_w[0..k] (gap_w[k] sits between atom k and k+1, so it
316    // belongs to any line containing both). natural(i,j) for atoms i..j and
317    // the gaps between them = (PA[j]-PA[i]) + (PG[j-1]-PG[i]).
318    let mut pa = vec![0.0f32; atoms.len() + 1];
319    for k in 0..atoms.len() {
320        pa[k + 1] = pa[k] + atom_w[k];
321    }
322    let mut pg = vec![0.0f32; atoms.len()];
323    for k in 0..gap_w.len() {
324        pg[k + 1] = pg[k] + gap_w[k];
325    }
326    let natural = |i: usize, j: usize| -> f32 {
327        let g = if j > 0 { pg[j - 1] - pg[i] } else { 0.0 };
328        (pa[j] - pa[i]) + g
329    };
330
331    let line_metrics = |lo: usize, hi: usize| {
332        let mut m = text.measure(&input[lo..hi], base);
333        m.width = measure(lo..hi);
334        m
335    };
336
337    // Run the DP independently over each run of atoms between forced (`\n`)
338    // breaks; each segment's final line gets the \parfillskip exemption.
339    let mut out: Vec<WrappedLine> = Vec::new();
340    let mut seg_start = 0usize;
341    while seg_start < atoms.len() {
342        let mut seg_end = seg_start;
343        while seg_end + 1 < atoms.len() && !gap_hard[seg_end] {
344            seg_end += 1;
345        }
346        // Segment = atoms[seg_start..=seg_end]; `na` atoms, local idx 0..=na-1.
347        let na = seg_end - seg_start + 1;
348        knuth_plass_segment(
349            seg_start,
350            na,
351            &atoms,
352            &natural,
353            &no_start,
354            &no_end,
355            em,
356            max_width,
357            &line_metrics,
358            &mut out,
359        );
360        seg_start = seg_end + 1;
361    }
362
363    out
364}

366/// Run the K-P DP over one hard-break-free segment of `na` atoms starting at
367/// global atom index `seg_start`, appending the resulting lines to `out`.
368///
369/// State `dp[j][c]` = best (cost, back_i, back_c) to lay out the first `j`
370/// local atoms such that the line ending at `j` is in fitness class `c`.
371/// `j` ranges `0..=na`; `c` ranges `0..NCLASS`. The class discretisation is
372/// what makes the DP finite — the continuous shortfall is carried only
373/// implicitly through the class.
374#[allow(clippy::too_many_arguments)]
375fn knuth_plass_segment(
376    seg_start: usize,
377    na: usize,
378    atoms: &[Atom],
379    natural: &impl Fn(usize, usize) -> f32,
380    no_start: &[bool],
381    no_end: &[bool],
382    em: f32,
383    max_width: f32,
384    line_metrics: &impl Fn(usize, usize) -> Metrics,
385    out: &mut Vec<WrappedLine>,
386) {
387    const NCLASS: usize = 4;
388    let inf = f32::INFINITY;
389    // dp[0] is the source: zero cost, no predecessor. Its four entries are
390    // identical; transitions treat i == 0 specially (no evenness demerit).
391    let mut dp: Vec<[(f32, usize, u8); NCLASS]> = vec![[(inf, 0, 0); NCLASS]; na + 1];
392    dp[0] = [(0.0, 0, 0); NCLASS];
393
394    // Scan i backwards from j-1: as i shrinks the line gains atoms, so width
395    // rises and shortfall falls. Overfull multi-atom lines let us bail early
396    // (smaller i only adds width); too-loose lines just get skipped (smaller
397    // i may bring the line back under the looseness ceiling).
398    for j in 1..=na {
399        // Kinsoku: a line may end at j only if atom j-1 may end a line and
400        // (unless j is the segment's final line) atom j may begin one.
401        // Leaving dp[j] at INF makes the DP span over j — the atoms around
402        // j join a longer line that ends at a legal break point.
403        if j < na && (no_end[seg_start + j - 1] || no_start[seg_start + j]) {
404            continue;
405        }
406        for i in (0..j).rev() {
407            let nat = natural(seg_start + i, seg_start + j);
408            let single = j - i == 1;
409            let is_last = j == na;
410            if nat > max_width && !single {
411                break;
412            }
413            let shortfall_em = ((max_width - nat).max(0.0)) / em;
414            // Final line: up to KP_LAST_LINE_FREE_EM of shortfall is free
415            // (≈ \parfillskip) — a short last line is correct, not bad.
416            let eff_em = if is_last {
417                (shortfall_em - KP_LAST_LINE_FREE_EM).max(0.0)
418            } else {
419                shortfall_em
420            };
421            if !is_last && !single && shortfall_em > KP_MAX_SHORTFALL_EM {
422                continue;
423            }
424            let class = kp_class(eff_em);
425            let badness = eff_em * eff_em * eff_em;
426
427            // Best predecessor over the previous line's class.
428            let (mut best, mut best_cprev) = (inf, 0u8);
429            if i == 0 {
430                best = 0.0;
431            } else {
432                for (cprev, &(pcost, _, _)) in dp[i].iter().enumerate() {
433                    if pcost >= inf {
434                        continue;
435                    }
436                    let even = if cprev == class {
437                        0.0
438                    } else {
439                        KP_CLASS_DEMERIT
440                    };
441                    let cost = pcost + even;
442                    if cost < best {
443                        best = cost;
444                        best_cprev = cprev as u8;
445                    }
446                }
447            }
448            if best >= inf {
449                continue;
450            }
451            let cand = best + badness;
452            if cand < dp[j][class].0 {
453                dp[j][class] = (cand, i, best_cprev);
454            }
455        }
456    }
457
458    // Recover the cheapest final state and walk the back-pointers.
459    let (mut end_c, mut end_cost) = (0usize, inf);
460    for (c, state) in dp[na].iter().enumerate() {
461        if state.0 < end_cost {
462            end_cost = state.0;
463            end_c = c;
464        }
465    }
466    let mut breaks: Vec<usize> = Vec::with_capacity(na + 1);
467    breaks.push(na);
468    let (mut cur_j, mut cur_c) = (na, end_c);
469    while cur_j > 0 {
470        let (_, back_i, back_c) = dp[cur_j][cur_c];
471        breaks.push(back_i);
472        cur_j = back_i;
473        if cur_j > 0 {
474            cur_c = back_c as usize;
475        }
476    }
477    breaks.reverse(); // [0, b1, ..., na]
478
479    for w in breaks.windows(2) {
480        let (li, lj) = (w[0], w[1]);
481        let glo = atoms[seg_start + li].start;
482        let ghi = atoms[seg_start + lj - 1].end;
483        out.push(WrappedLine {
484            lo: glo,
485            hi: ghi,
486            metrics: line_metrics(glo, ghi),
487        });
488    }
489}

491/// Fitness class 0..3 for a line whose effective shortfall is `eff_em` EM
492/// (tight / decent / loose / very-loose). Used by [`knuth_plass_segment`].
493fn kp_class(eff_em: f32) -> usize {
494    if eff_em < KP_FITNESS_EM[0] {
495        0
496    } else if eff_em < KP_FITNESS_EM[1] {
497        1
498    } else if eff_em < KP_FITNESS_EM[2] {
499        2
500    } else {
501        3
502    }
503}
504
505/// Fitness-class boundaries in EM of shortfall (max_width − natural width).
506/// Four classes (the standard K-P count) — discretising shortfall into
507/// classes is what keeps the DP finite: the state carries the previous
508/// line's class, not its continuous shortfall.
509const KP_FITNESS_EM: [f32; 3] = [0.5, 1.5, 3.0];
510
511/// Demerit added when two adjacent lines land in different fitness classes.
512/// Scaled to be comparable to a line's badness (≈ shortfall_em³), so the
513/// optimiser trades line badness against evenness rather than one
514/// dominating. Tunable.
515const KP_CLASS_DEMERIT: f32 = 10.0;
516
517/// A non-final, non-single-atom line whose shortfall exceeds this (in EM) is
518/// "too loose" to consider — rules out absurd breaks and bounds the active
519/// set. Single-atom lines and each segment's final line are exempt.
520const KP_MAX_SHORTFALL_EM: f32 = 6.0;
521
522/// Free shortfall granted to a segment's final line (≈ TeX's
523/// `\parfillskip`): a short last line is correct, not bad, so up to this
524/// much costs nothing. Beyond it the line is penalised normally — which is
525/// what discourages stranding a single word on the final line.
526const KP_LAST_LINE_FREE_EM: f32 = 2.0;
527
528/// Atom count above which K-P falls back to greedy. K-P is O(n²·C)
529/// (C = [`kp_class`]'s 4 classes); past this a megabyte paste could drop a
530/// frame. Well beyond any realistic paragraph.
531const KP_GREEDY_FALLBACK_ATOMS: usize = 3000;
532
533/// A maximal run of source bytes that the wrap algorithm treats as atomic
534/// (cannot be split). See [`wrap`].
535#[derive(Debug, Clone, Copy, PartialEq, Eq)]
536struct Atom {
537    start: usize,
538    end: usize,
539}

541/// Split `input` into wrap atoms. Whitespace and `\n` are *separators* (not
542/// atoms); they define break opportunities between atoms. Atoms are:
543///
544/// - a maximal run of non-whitespace, non-CJK characters (a Latin "word",
545///   including digits, punctuation, emoji ZWJ sequences by codepoint), or
546/// - a single CJK ideograph (which is its own atom so CJK text wraps without
547///   spaces),
548///
549/// Adjacent atoms imply a break opportunity between them.
550fn split_atoms(input: &str) -> Vec<Atom> {
551    let mut atoms = Vec::new();
552    let mut chars = input.char_indices().peekable();
553
554    while let Some((i, ch)) = chars.next() {
555        if ch == '\n' || ch.is_whitespace() {
556            // Separator — skip. Whitespace runs collapse together because
557            // each whitespace char takes this branch.
558            continue;
559        }
560
561        let start = i;
562        if is_cjk(ch) {
563            // Single CJK char is its own atom.
564            atoms.push(Atom {
565                start,
566                end: i + ch.len_utf8(),
567            });
568        } else {
569            // Read a maximal run of non-whitespace, non-CJK chars.
570            let mut end = i + ch.len_utf8();
571            while let Some(&(_, next_ch)) = chars.peek() {
572                if next_ch == '\n' || next_ch.is_whitespace() || is_cjk(next_ch) {
573                    break;
574                }
575                end += next_ch.len_utf8();
576                chars.next();
577            }
578            atoms.push(Atom { start, end });
579        }
580    }
581
582    atoms
583}

585/// Whether `ch` is a CJK ideograph or related script where line-wrap should
586/// be allowed at every character. Covers CJK Unified, CJK Ext A/B/C/D/E/F/G,
587/// Hiragana, Katakana, CJK symbols/punctuation, Hangul, and full-width
588/// forms. Not exhaustive, but covers the common case: text in CJK locales
589/// wraps character-by-character rather than at whitespace.
590fn is_cjk(ch: char) -> bool {
591    matches!(ch as u32,
592        0x1100..=0x11FF   // Hangul Jamo
593        | 0x2E80..=0x2EFF // CJK Radicals Supplement
594        | 0x2F00..=0x2FDF // Kangxi Radicals
595        | 0x3000..=0x303F // CJK Symbols and Punctuation
596        | 0x3040..=0x309F // Hiragana
597        | 0x30A0..=0x30FF // Katakana
598        | 0x3100..=0x312F // Bopomofo
599        | 0x3130..=0x318F // Hangul Compatibility Jamo
600        | 0x3400..=0x4DBF // CJK Unified Ideographs Extension A
601        | 0x4E00..=0x9FFF // CJK Unified Ideographs
602        | 0xA960..=0xA97F // Hangul Jamo Extended-A
603        | 0xAC00..=0xD7AF // Hangul Syllables
604        | 0xD7B0..=0xD7FF // Hangul Jamo Extended-B
605        | 0xF900..=0xFAFF // CJK Compatibility Ideographs
606        | 0xFF00..=0xFFEF // Halfwidth and Full-width Forms
607        | 0x1F300..=0x1FAFF // Symbols & Pictographs + CJK symbols (broad emoji)
608        | 0x20000..=0x2A6DF // CJK Unified Ideographs Extension B
609        | 0x2A700..=0x2B73F // CJK Ext C
610        | 0x2B740..=0x2B81F // CJK Ext D
611        | 0x2B820..=0x2CEAF // CJK Ext E
612        | 0x2CEB0..=0x2EBEF // CJK Ext F
613        | 0x30000..=0x3134F // CJK Ext G
614    )
615}

617/// Whether `atom`'s first codepoint is CJK — the atom's script class for the
618/// inter-script auto-space rule in [`wrap`]. Derived from the text rather
619/// than stored on [`Atom`], so the struct stays minimal and the rule reads
620/// its input from the single source of truth (the codepoint), matching how
621/// the C shaper derives the same fact from `run->script`.
622fn atom_is_cjk(input: &str, atom: Atom) -> bool {
623    input[atom.start..].chars().next().is_some_and(is_cjk)
624}
625
626/// The `(width, is_hard_break)` of the gap between atoms `idx - 1` and
627/// `idx` — i.e. the source bytes `atoms[idx-1].end .. atoms[idx].start`.
628///
629/// The width includes the inter-script auto-space when the two atoms cross
630/// the CJK line and no separator already covers the boundary. This is the
631/// **single source of truth** for that rule: both the greedy
632/// ([`wrap_with`]) and optimal ([`wrap_optimal_with`]) strategies call it,
633/// so neither can drift from the ¼-em gap the shaper inserts.
634fn gap_cost(input: &str, atoms: &[Atom], idx: usize, space_w: f32, size_px: f32) -> (f32, bool) {
635    let gap = &input[atoms[idx - 1].end..atoms[idx].start];
636    let hard = gap.contains('\n');
637    let mut w = space_w * gap.bytes().filter(|&b| b != b'\n').count() as f32;
638    // `is_cjk` is a superset of the shaper's script test (it also covers
639    // emoji / full-width), so this only ever over-estimates relative to the
640    // shaper — the safe direction for line-break accounting.
641    if w == 0.0 && atom_is_cjk(input, atoms[idx - 1]) != atom_is_cjk(input, atoms[idx]) {
642        w += size_px * INTERSCRIPT_GAP_EM;
643    }
644    (w, hard)
645}

647/// CJK punctuation that must not **begin** a line: closing brackets and
648/// stop punctuation (。，、）》」…). JLREQ cl-02 / cl-06 / cl-07. A break
649/// immediately before one would strand it at a line start, so kinsoku shori
650/// forbids such breaks (the char is pulled back onto the previous line).
651fn is_no_start(ch: char) -> bool {
652    matches!(
653        ch,
654        '〉' | '》'
655            | '」'
656            | '』'
657            | '】'
658            | '〕'
659            | '〗'
660            | '〙'
661            | '〟'
662            | '）'
663            | '］'
664            | '｝'
665            | '｠'
666            | '、'
667            | '。'
668            | '，'
669            | '．'
670            | '：'
671            | '；'
672            | '！'
673            | '？'
674    )
675}
676
677/// CJK opening brackets that must not **end** a line: （（「『…. JLREQ
678/// cl-01. A break immediately after one would leave it dangling at a line
679/// end, separated from what it opens, so kinsoku forbids such breaks (the
680/// bracket is pushed onto the next line).
681fn is_no_end(ch: char) -> bool {
682    matches!(
683        ch,
684        '〈' | '《' | '「' | '『' | '【' | '〔' | '〖' | '〚' | '〝' | '（' | '［' | '｛' | '｟'
685    )
686}

688/// Whether `atom`'s first character is no-start (must not begin a line).
689fn atom_no_start(input: &str, atom: Atom) -> bool {
690    input[atom.start..].chars().next().is_some_and(is_no_start)
691}
692
693/// Whether `atom`'s last character is no-end (must not end a line).
694fn atom_no_end(input: &str, atom: Atom) -> bool {
695    input[..atom.end].chars().last().is_some_and(is_no_end)
696}

1104        // Four atoms whose widths (via the callback) are 50, 50, 50, 30 px,
1105        // at a column that fits three 50s but not the 30 as well. Greedy
1106        // packs three onto line 1 and strands the 30 alone on line 2.
1107        // Knuth-Plass moves one 50 down so both lines are balanced — the
1108        // visible benefit of whole-paragraph optimisation over greedy fill.
1109        let Some(engine) = engine() else { return };
1110        let st = style();
1111        let text = "aaaaa aaaaa aaaaa aaa"; // byte widths 5,5,5,3
1112        let px = |r: std::ops::Range<usize>| (r.end - r.start) as f32 * 10.0; // -> 50,50,50,30
1113        let space_w = engine.measure(" ", &st).width;
1114        // Fits three 50s + two gaps; the fourth atom tips it over.
1115        let max_w = 3.0 * 50.0 + 2.0 * space_w + 10.0;
1116
1117        let greedy = wrap_with(&engine, text, &st, max_w, px);
1118        let optimal = wrap_optimal_with(&engine, text, &st, max_w, px);
1119
1120        // Both produce two lines; the difference is distribution.
1121        assert_eq!(greedy.len(), 2);
1122        assert_eq!(optimal.len(), 2);
1123        // Greedy's first line runs to the end of the third atom (byte 17);
1124        // optimal breaks earlier, moving one atom onto line 2.
1125        assert_eq!(greedy[0].hi, 17, "greedy fills line 1 with three atoms");
1126        assert!(
1127            optimal[0].hi < greedy[0].hi,
1128            "K-P should rebalance (break line 1 earlier), got hi={}",
1129            optimal[0].hi
1130        );
1131        // Contract still holds.
1132        assert_eq!(optimal[0].lo, 0);
1133        assert_eq!(optimal.last().unwrap().hi, text.len());

1183    fn wrap_with_uses_caller_supplied_measure() {
1184        // A caller-supplied measure that reports each atom as 10px per
1185        // source byte — unrelated to the real glyph widths. wrap_with must
1186        // wrap against THAT width, proving atom widths route through
1187        // `measure` rather than the raw-source shape.
1188        let Some(engine) = engine() else { return };
1189        let st = style();
1190        let text = "alpha beta gamma delta epsilon";
1191        // Raw wrap at a wide column: one line.
1192        assert_eq!(wrap(&engine, text, &st, 10000.0).len(), 1);
1193        // Custom measure: 10px/byte. The string is 30 bytes => 300px, so a
1194        // 100px column must wrap to several lines.
1195        let measured = wrap_with(&engine, text, &st, 100.0, |r| {
1196            (r.end - r.start) as f32 * 10.0
1197        });
1198        assert!(
1199            measured.len() > 1,
1200            "wrap_with must honour the caller-supplied measure"
1201        );
1202        // The WrappedLine contract is unchanged: lines tile the input.
1203        assert_eq!(measured[0].lo, 0);
1204        for w in measured.windows(2) {
1205            assert!(w[0].hi <= w[1].lo);
1206        }
1207        assert_eq!(measured.last().unwrap().hi, text.len());
1208        // And the returned metric width is the caller's drawn width, not a
1209        // re-shape of the raw source.
1210        let first = &measured[0];
1211        let raw_width = engine.measure(&text[first.lo..first.hi], &st).width;
1212        assert!(
1213            (first.metrics.width - (first.hi - first.lo) as f32 * 10.0).abs() < 0.01,
1214            "metrics.width must come from `measure`, got {} vs raw {}",
1215            first.metrics.width,
1216            raw_width
1217        );
