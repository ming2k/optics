# Observation-plane attribution (ADR-0129, ADR-0130). Off by default: a
# release build must not carry the name/site machinery at all.
option('observe_attribution', type : 'boolean', value : false,
       description : 'Compile object-name and creation-site attribution into the ledger')

# Tracy profiler instrumentation (ADR-0138). Off by default: a release
# build carries no Tracy runtime or C++ dependencies.
option('tracy', type : 'boolean', value : false,
       description : 'Compile Tracy profiler instrumentation into the libraries (ADR-0138)')

option('text',   type : 'boolean', value : true,  description : 'Build the text shaping library (ADR-0106)')
option('motion', type : 'boolean', value : true,  description : 'Build the header-only motion vocabulary library (ADR-0077, ADR-0106)')

option('examples', type : 'boolean', value : false, description : 'Build all C examples across all modules')
option('tests',    type : 'boolean', value : false, description : 'Build all C tests across all modules')
option('fuzz',     type : 'boolean', value : false, description : 'Build fuzz harnesses (links libFuzzer when -Db_lundef=false + LIBFUZZER=1 env var)')
